const SPREADSHEET_ID = '14tJtNr-Nx6tJolgoiBsxX5rRo1_hRADK9Evr8rA3low';
const SHEET_NAME     = 'Kunjungan';
const TIMEZONE       = 'Asia/Makassar'; // WITA — Sulawesi Tenggara

function doGet(e)  { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  try {
    const params = (e && e.parameter) || {};
    const action = params.action || 'stats';

    if (action === 'visit')   recordVisit(params);
    if (action === 'rebuild') rebuildStats();

    const stats = getStats();
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, data: stats }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message || String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/* ============================================================
   SHEET MANAGEMENT
============================================================ */
function getSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    const headers = ['Timestamp', 'Tanggal', 'Bulan', 'Tahun',
                     'VisitorID', 'UserAgent', 'Referrer', 'Negara'];
    sheet.getRange(1, 1, 1, headers.length)
      .setValues([headers])
      .setFontWeight('bold')
      .setBackground('#0f4c81')
      .setFontColor('#ffffff');
    sheet.setFrozenRows(1);
    sheet.setColumnWidth(1, 170);
    sheet.setColumnWidth(2, 110);
    sheet.setColumnWidth(3, 95);
    sheet.setColumnWidth(4, 75);
    sheet.setColumnWidth(5, 230);
    sheet.setColumnWidth(6, 220);
    sheet.setColumnWidth(7, 220);
    sheet.setColumnWidth(8, 100);
  }

  // KUNCI PERBAIKAN: paksa kolom B, C, D jadi format teks
  // agar Google Sheets TIDAK mengonversi "2026-06-01" → Date
  sheet.getRange('B:D').setNumberFormat('@');

  return sheet;
}

/* ============================================================
   HELPER: STRING TANGGAL KONSISTEN
============================================================ */
function todayStr() { return Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd'); }
function monthStr() { return Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM'); }
function yearStr()  { return Utilities.formatDate(new Date(), TIMEZONE, 'yyyy'); }

function formatLongDate(d) {
  const days   = ['Minggu','Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];
  const months = ['Januari','Februari','Maret','April','Mei','Juni',
                  'Juli','Agustus','September','Oktober','November','Desember'];
  return days[d.getDay()] + ', ' + d.getDate() + ' ' +
         months[d.getMonth()] + ' ' + d.getFullYear();
}

function formatMonthLabel(d) {
  const months = ['Januari','Februari','Maret','April','Mei','Juni',
                  'Juli','Agustus','September','Oktober','November','Desember'];
  return months[d.getMonth()] + ' ' + d.getFullYear();
}

/* ============================================================
   RECORD VISIT
============================================================ */
function recordVisit(params) {
  const vid = params.vid || '';
  const now = new Date();

  // Dedup: 1 Visitor ID = 1 kunjungan per hari
  if (vid) {
    const cache = CacheService.getScriptCache();
    const key   = 'v_' + todayStr() + '_' + vid.substring(0, 40);
    if (cache.get(key)) return;
    cache.put(key, '1', 21600); // 6 jam
  }

  const sheet = getSheet();
  const row   = sheet.getLastRow() + 1;

  sheet.getRange(row, 1, 1, 8).setValues([[
    now,
    todayStr(),       // B — yyyy-MM-dd
    monthStr(),       // C — yyyy-MM
    yearStr(),        // D — yyyy
    vid,
    params.ua || '',
    params.ref || '',
    params.country || ''
  ]]);

  // Pastikan format teks setelah menulis (anti auto-convert)
  sheet.getRange(row, 2, 1, 3).setNumberFormat('@');
}

/* ============================================================
   GET STATS — PERBAIKAN UTAMA
============================================================ */
function getStats() {
  const sheet   = getSheet();
  const lastRow = sheet.getLastRow();

  const today = todayStr();   // "2026-09-29"
  const month = monthStr();   // "2026-09"
  const year  = yearStr();    // "2026"

  const now = new Date();

  const stats = {
    today: 0,
    month: 0,
    year:  0,
    total: 0,
    // Label periode yang dibandingkan (untuk ditampilkan di frontend)
    todayLabel: formatLongDate(now),     // "Senin, 29 September 2026"
    monthLabel: formatMonthLabel(now),   // "September 2026"
    yearLabel:  year,                    // "2026"
    todayKey:   today,                   // "2026-09-29"
    monthKey:   month,                   // "2026-09"
    yearKey:    year,
    updated:    new Date().toISOString(),
    timezone:   TIMEZONE
  };

  if (lastRow < 2) return stats;
  stats.total = lastRow - 1;

  // KUNCI PERBAIKAN: gunakan getDisplayValues() — SELALU string
  // (getValues() bisa return Date object → perbandingan gagal)
  const data = sheet.getRange(2, 2, lastRow - 1, 3).getDisplayValues();

  for (let i = 0; i < data.length; i++) {
    const tgl = String(data[i][0] || '').trim(); // kolom B
    const bln = String(data[i][1] || '').trim(); // kolom C
    const thn = String(data[i][2] || '').trim(); // kolom D

    // Perbandingan string ke string (keduanya format yyyy-MM-dd / yyyy-MM / yyyy)
    if (tgl === today) stats.today++;
    if (bln === month) stats.month++;
    if (thn === year)  stats.year++;
  }

  return stats;
}

/* ============================================================
   REBUILD — perbaiki data lama yang formatnya salah
   Jalankan manual dari editor Apps Script jika perlu.
============================================================ */
function rebuildStats() {
  const sheet   = getSheet();
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const range = sheet.getRange(2, 1, lastRow - 1, 4);
  const data  = range.getValues();

  for (let i = 0; i < data.length; i++) {
    const ts = data[i][0];
    let d;

    if (ts instanceof Date) {
      d = ts;
    } else {
      d = new Date(ts);
    }
    if (isNaN(d.getTime())) continue;

    data[i][1] = Utilities.formatDate(d, TIMEZONE, 'yyyy-MM-dd');
    data[i][2] = Utilities.formatDate(d, TIMEZONE, 'yyyy-MM');
    data[i][3] = Utilities.formatDate(d, TIMEZONE, 'yyyy');
  }

  range.setValues(data);
  sheet.getRange('B:D').setNumberFormat('@');
  Logger.log('✅ Rebuild selesai untuk ' + data.length + ' baris.');
}

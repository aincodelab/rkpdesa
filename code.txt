const SPREADSHEET_ID = '14tJtNr-Nx6tJolgoiBsxX5rRo1_hRADK9Evr8rA3low';
const SHEET_NAME     = 'Kunjungan';

function doGet(e)  { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }

function handleRequest(e) {
  try {
    const params = (e && e.parameter) || {};
    const action = params.action || 'stats';

    if (action === 'visit') recordVisit(params);

    const stats = getStats();
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, data: stats }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message || String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    const headers = ['Timestamp', 'Tanggal', 'Bulan', 'Tahun', 'VisitorID', 'UserAgent', 'Referrer', 'Negara'];
    sheet.getRange(1, 1, 1, headers.length)
      .setValues([headers])
      .setFontWeight('bold')
      .setBackground('#0f4c81')
      .setFontColor('#ffffff');
    sheet.setFrozenRows(1);
    sheet.setColumnWidth(1, 170);
    sheet.setColumnWidth(2, 100);
    sheet.setColumnWidth(3, 90);
    sheet.setColumnWidth(4, 70);
    sheet.setColumnWidth(5, 230);
    sheet.setColumnWidth(6, 220);
    sheet.setColumnWidth(7, 220);
    sheet.setColumnWidth(8, 100);
  }
  return sheet;
}

function recordVisit(params) {
  const vid = params.vid || '';

  const now      = new Date();
  const tz       = Session.getScriptTimeZone();
  const dateStr  = Utilities.formatDate(now, tz, 'yyyy-MM-dd');
  const monthStr = Utilities.formatDate(now, tz, 'yyyy-MM');
  const yearStr  = Utilities.formatDate(now, tz, 'yyyy');

  // Deduplikasi: 1 visitor ID = 1 kunjungan per hari (cache 6 jam)
  if (vid) {
    const cache = CacheService.getScriptCache();
    const key   = 'v_' + dateStr + '_' + vid.substring(0, 40);
    if (cache.get(key)) return;
    cache.put(key, '1', 21600);
  }

  getSheet().appendRow([
    now,
    dateStr,
    monthStr,
    yearStr,
    vid,
    params.ua || '',
    params.ref || '',
    params.country || ''
  ]);
}

function getStats() {
  const sheet   = getSheet();
  const lastRow = sheet.getLastRow();

  const tz    = Session.getScriptTimeZone();
  const now   = new Date();
  const today = Utilities.formatDate(now, tz, 'yyyy-MM-dd');
  const month = Utilities.formatDate(now, tz, 'yyyy-MM');
  const year  = Utilities.formatDate(now, tz, 'yyyy');

  const stats = { today: 0, month: 0, year: 0, total: 0, updated: new Date().toISOString() };
  if (lastRow < 2) return stats;

  stats.total = lastRow - 1;

  const data = sheet.getRange(2, 2, lastRow - 1, 3).getValues();
  for (let i = 0; i < data.length; i++) {
    if (data[i][0] === today) stats.today++;
    if (data[i][1] === month) stats.month++;
    if (data[i][2] === year)  stats.year++;
  }
  return stats;
}

function testSetup() {
  const sheet = getSheet();
  Logger.log('Sheet    : ' + sheet.getName());
  Logger.log('Baris    : ' + sheet.getLastRow());
  Logger.log('Stats    : ' + JSON.stringify(getStats()));
}
